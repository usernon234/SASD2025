<h2>Dead Code</h2>
<h4>Dead Code คือส่วนของซอร์สโค้ดที่เขียนไว้แต่ ไม่มีวันถูกเรียกใช้งาน หรือ ไม่มีผลลัพธ์ต่อการทำงานของโปรแกรม ในขณะที่แอปพลิเคชันทำงานจริงไม่ว่าจะรันกี่ครั้งก็ตาม 
เช่น ฟังก์ชันที่ไม่มีใครเรียก ตัวแปรที่ประกาศไว้แต่ไม่เคยใช้ เงื่อนไขที่ไม่มีทางเป็นจริง</h4>
<h4>ตัวอย่างโค้ดที่เกิด Dead Code</h4>

    const taxRate = 0.07;

<h4>ตัวอย่างโค้ดที่ Refactoring แล้ว</h4>
     
    function calculateFinalPrice(price, discount) {
    const taxRate = 0.07;
    // ลบ serviceCharge ออกเพราะไม่ได้ใช้
    
    let finalPrice = price - discount;

    if (finalPrice < 0) {
        return 0;
    }

    return finalPrice + (finalPrice * taxRate);
    // ลบ console.log ที่อยู่หลัง return ออก
    }
